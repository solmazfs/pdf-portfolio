\newcommand{\PortfolioCoverImage}{%
    \savebox{\PortfolioCoverImageBox}{\includegraphics{$cover-image$}}%
    \begin{tikzpicture}[remember picture,overlay]
        \begin{scope}
            \clip ([xshift=20mm,yshift=20mm]current page.south west)
            rectangle ([xshift=140mm,yshift=180mm]current page.south west);
            \ifdim\wd\PortfolioCoverImageBox<.75\ht\PortfolioCoverImageBox
                \node[anchor=center,inner sep=0pt]
                at ([xshift=80mm,yshift=100mm]current page.south west)
                {\includegraphics[width=12cm]{$cover-image$}};
            \else
                \node[anchor=center,inner sep=0pt]
                at ([xshift=80mm,yshift=100mm]current page.south west)
                {\includegraphics[height=16cm]{$cover-image$}};
            \fi
        \end{scope}
    \end{tikzpicture}%
}
